// --- Historique des evenements de mise en securite vent ---
#define NB_EVENEMENTS_VENT 8
String histHeure[NB_EVENEMENTS_VENT];
float  histVitesse[NB_EVENEMENTS_VENT];
String histCause[NB_EVENEMENTS_VENT];
int histIndex = 0;
int histRempli = 0;

#define OTA_PASSWORD "MASQUE_VOIR_SECRETS_H"  // mot de passe pour flasher via WiFi (optionnel mais recommande)
Preferences prefs;                   // stockage persistant NVS pour l'historique vent

void sauvegarderHistoriqueVent() {
  // Sauvegarde l'historique du vent en mémoire flash (NVS)
  // Format : chaque événement = "heure,vitesse,cause|" concaténés
  String data = "";
  for (int i = 0; i < histRempli; i++) {
    int idx = (histIndex - histRempli + i + NB_EVENEMENTS_VENT) % NB_EVENEMENTS_VENT;
    data += histHeure[idx] + "," + String(histVitesse[idx], 1) + "," + histCause[idx] + "|";
  }
  prefs.putString("histVent", data);
}

void chargerHistoriqueVent() {
  // Charge l'historique du vent depuis la mémoire flash (NVS)
  String data = prefs.getString("histVent", "");
  if (data.length() == 0) return; // aucun historique sauvegardé
  
  histIndex = 0;
  histRempli = 0;
  
  int pos = 0;
  while (pos < data.length()) {
    int delim1 = data.indexOf(',', pos);
    int delim2 = data.indexOf(',', delim1 + 1);
    int delim3 = data.indexOf('|', delim2 + 1);
    if (delim1 == -1 || delim2 == -1 || delim3 == -1) break;
    
    String h = data.substring(pos, delim1);
    float v = data.substring(delim1 + 1, delim2).toFloat();
    String c = data.substring(delim2 + 1, delim3);
    
    histHeure[histIndex] = h;
    histVitesse[histIndex] = v;
    histCause[histIndex] = c;
    histIndex = (histIndex + 1) % NB_EVENEMENTS_VENT;
    if (histRempli < NB_EVENEMENTS_VENT) histRempli++;
    
    pos = delim3 + 1;
  }
  histIndex = histRempli % NB_EVENEMENTS_VENT; // remet l'index au bon endroit
  Serial.printf("Historique vent charge : %d evenements.\n", histRempli);
}

// --- Tendance du vent (augmentation / diminution) ---
#define NB_ECHANTILLONS_TENDANCE 6      // 6 x FENETRE_VENT_MS (5s) = 30s de fenetre d'analyse
#define SEUIL_TENDANCE_KMH        2.0   // ecart mini pour ne pas declencher sur du bruit
float bufferVent[NB_ECHANTILLONS_TENDANCE];
int bufferVentIndex = 0;
int bufferVentRempli = 0;
String tendanceVent = "stable"; // "up" / "down" / "stable"

String calculerTendanceVent() {
  if (bufferVentRempli < 4) return "stable"; // pas assez d'historique pour juger

  int moitie = bufferVentRempli / 2;
  float sommeAncien = 0, sommeRecent = 0;
  int nAncien = 0, nRecent = 0;

  for (int i = 0; i < bufferVentRempli; i++) {
    int idx = (bufferVentIndex - bufferVentRempli + i + NB_ECHANTILLONS_TENDANCE) % NB_ECHANTILLONS_TENDANCE;
    if (i < moitie) { sommeAncien += bufferVent[idx]; nAncien++; }
    else            { sommeRecent += bufferVent[idx]; nRecent++; }
  }

  float diff = (sommeRecent / nRecent) - (sommeAncien / nAncien);
  if (diff > SEUIL_TENDANCE_KMH)  return "up";
  if (diff < -SEUIL_TENDANCE_KMH) return "down";
  return "stable";
}

// Enregistre un evenement de mise en securite (moyenne ou rafale) dans l'historique circulaire
// (definie plus bas, apres la declaration de rtc/rtcOK — voir juste apres RTC_DS3231 rtc;)

void IRAM_ATTR compterImpulsionVent() {
  unsigned long maintenant = millis();
  if (maintenant - dernierImpulsionVent > DEBOUNCE_VENT_MS) {
    nbImpulsionsVent++;
    nbImpulsionsRafale++;
    dernierImpulsionVent = maintenant;
  }
}

void setupAnemometre() {
  pinMode(PIN_ANEMOMETRE, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_ANEMOMETRE), compterImpulsionVent, FALLING);
}

void setupOTA() {
  // Configure le flashage Over-The-Air (via WiFi)
  ArduinoOTA.setHostname("tracker-solaire");
  ArduinoOTA.setPassword(OTA_PASSWORD);
  
  ArduinoOTA.onStart([]() {
    Serial.println("\n=== OTA : debut du flashage ===");
  });
  
  ArduinoOTA.onEnd([]() {
    Serial.println("\n=== OTA : flashage termine, redemarrage en cours ===");
  });
  
  ArduinoOTA.onProgress([](unsigned int progress, unsigned int total) {
    Serial.printf("OTA Progress: %u%%\r", (progress / (total / 100)));
  });
  
  ArduinoOTA.onError([](ota_error_t error) {
    Serial.printf("OTA Error[%u]: ", error);
    if (error == OTA_AUTH_ERROR) Serial.println("Auth Failed");
    else if (error == OTA_BEGIN_ERROR) Serial.println("Begin Failed");
    else if (error == OTA_CONNECT_ERROR) Serial.println("Connect Failed");
    else if (error == OTA_RECEIVE_ERROR) Serial.println("Receive Failed");
    else if (error == OTA_END_ERROR) Serial.println("End Failed");
  });
  
  ArduinoOTA.begin();
  Serial.println("OTA ready (port 3232, password: " OTA_PASSWORD ")");
}

void updateVent() {
  if (millis() - lastMesureVent >= FENETRE_VENT_MS) {
    noInterrupts();
    unsigned long imp = nbImpulsionsVent;
    nbImpulsionsVent = 0;
    interrupts();

    float impParSeconde = imp / (FENETRE_VENT_MS / 1000.0);
    float nouvelleVitesse = impParSeconde * FACTEUR_KMH;
    if (nouvelleVitesse > 0 || imp == 0) vitesseVentKmh = nouvelleVitesse;

    bufferVent[bufferVentIndex] = vitesseVentKmh;
    bufferVentIndex = (bufferVentIndex + 1) % NB_ECHANTILLONS_TENDANCE;
    if (bufferVentRempli < NB_ECHANTILLONS_TENDANCE) bufferVentRempli++;
    tendanceVent = calculerTendanceVent();

    // Palier intermediaire : amplitude de mouvement reduite (avant le seuil de securite)
    if (vitesseVentKmh >= SEUIL_VENT_REDUIT && !modeVentReduit) {
      modeVentReduit = true;
      Serial.printf("Vent modere (%.1f km/h) : amplitude de mouvement reduite.\n", vitesseVentKmh);
    }
    if (vitesseVentKmh < SEUIL_VENT_REDUIT * 0.8) {
      modeVentReduit = false; // desactivation avec hysteresis
    }

    // Mise en securite sur vent moyen (fenetre 5s)
    if (vitesseVentKmh >= SEUIL_VENT_SECURITE && !securiteVentActive) {
      securiteVentActive = true;
      Serial.printf("ALERTE VENT MOYEN : %.1f km/h - retour a plat automatique !\n", vitesseVentKmh);
      enregistrerEvenementVent(vitesseVentKmh, "moyenne");
    }
    if (vitesseVentKmh < SEUIL_VENT_SECURITE * 0.8) {
      securiteVentActive = false; // desactivation avec hysteresis
    }

    lastMesureVent = millis();
  }
}

// Detection de rafale : fenetre courte (2s) independante de la moyenne 5s, pour
// declencher la securite meme si un pic bref est lisse par la moyenne plus longue.
void updateRafale() {
  if (millis() - lastMesureRafale >= GUST_WINDOW_MS) {
    noInterrupts();
    unsigned long imp = nbImpulsionsRafale;
    nbImpulsionsRafale = 0;
    interrupts();

    float impParSeconde = imp / (GUST_WINDOW_MS / 1000.0);
    vitesseRafaleKmh = impParSeconde * FACTEUR_KMH;

    if (vitesseRafaleKmh >= SEUIL_RAFALE_SECURITE && !securiteVentActive) {
      securiteVentActive = true;
      Serial.printf("ALERTE RAFALE : %.1f km/h (pic %ds) - retour a plat automatique !\n",
                    vitesseRafaleKmh, GUST_WINDOW_MS / 1000);
      enregistrerEvenementVent(vitesseRafaleKmh, "rafale");
    }

    lastMesureRafale = millis();
  }
}

Adafruit_BMP280 bmp;
RTC_DS3231 rtc;
bool bmeOK = false;
bool rtcOK = false;

// Enregistre un evenement de mise en securite (moyenne ou rafale) dans l'historique circulaire
void enregistrerEvenementVent(float vitesse, const char* cause) {
  String heureStr = "??:??:??";
  if (rtcOK) {
    DateTime now = rtc.now();
    char buf[9];
    snprintf(buf, sizeof(buf), "%02d:%02d:%02d", now.hour(), now.minute(), now.second());
    heureStr = String(buf);
  }
  histHeure[histIndex]   = heureStr;
  histVitesse[histIndex] = vitesse;
  histCause[histIndex]   = String(cause);
  histIndex = (histIndex + 1) % NB_EVENEMENTS_VENT;
  if (histRempli < NB_EVENEMENTS_VENT) histRempli++;
  
  // Sauvegarde l'historique immediatement en mémoire flash (NVS)
  sauvegarderHistoriqueVent();
}

bool setupCapteursI2C() {
  Wire.begin(PIN_SDA, PIN_SCL);
  Wire.setClock(50000); // vitesse reduite, plus tolerante au bruit electrique
  delay(100); // laisse le bus/les capteurs se stabiliser

  for (int essai = 0; essai < 3 && !bmeOK; essai++) {
    bmeOK = bmp.begin(0x76, 0x58); // chip ID 0x58 = clone BMP280 HW-611
    if (!bmeOK) bmeOK = bmp.begin(0x77, 0x58);
    if (!bmeOK) delay(100);
  }
  if (bmeOK) {
    bmp.setSampling(Adafruit_BMP280::MODE_NORMAL,
                    Adafruit_BMP280::SAMPLING_X2,
                    Adafruit_BMP280::SAMPLING_X16,
                    Adafruit_BMP280::FILTER_X16,
                    Adafruit_BMP280::STANDBY_MS_500);
  }
  Wire.begin(PIN_SDA, PIN_SCL); // re-force les broches
  if (bmeOK) Serial.println("BMP280 detecte.");
  else Serial.println("BMP280 NON detecte (verifie le cablage/adresse).");

  rtcOK = rtc.begin(&Wire);
  Wire.begin(PIN_SDA, PIN_SCL); // re-force les broches (rtc.begin peut les reinitialiser)
  if (rtcOK) Serial.println("RTC DS3231 detectee.");
  else Serial.println("RTC NON detectee (verifie le cablage).");

  return bmeOK || rtcOK;
}

// Regle la RTC depuis un serveur NTP (appele une fois si le WiFi est up)
void synchroniserRTCDepuisNTP() {
  if (!rtcOK || WiFi.status() != WL_CONNECTED) return;

  configTime(0, 0, "pool.ntp.org", "time.google.com");
  struct tm timeinfo;
  if (getLocalTime(&timeinfo, 8000)) {
    rtc.adjust(DateTime(timeinfo.tm_year + 1900, timeinfo.tm_mon + 1, timeinfo.tm_mday,
                         timeinfo.tm_hour, timeinfo.tm_min, timeinfo.tm_sec));
    Serial.println("RTC synchronisee via NTP (UTC).");
  } else {
    Serial.println("Echec synchro NTP, la RTC garde son heure actuelle.");
  }
}

void setupMoteurs() {
  ledcAttach(RPWM_INC, PWM_FREQ, PWM_RES);
  ledcAttach(LPWM_INC, PWM_FREQ, PWM_RES);
  ledcAttach(RPWM_AZ, PWM_FREQ, PWM_RES);
  ledcAttach(LPWM_AZ, PWM_FREQ, PWM_RES);
  ledcWrite(RPWM_INC, 0);
  ledcWrite(LPWM_INC, 0);
  ledcWrite(RPWM_AZ, 0);
  ledcWrite(LPWM_AZ, 0);
}

void driveInclinaison(int8_t sens, uint8_t vitesse) {
  if (sens > 0)      { ledcWrite(RPWM_INC, vitesse); ledcWrite(LPWM_INC, 0); }
  else if (sens < 0) { ledcWrite(RPWM_INC, 0); ledcWrite(LPWM_INC, vitesse); }
  else               { ledcWrite(RPWM_INC, 0); ledcWrite(LPWM_INC, 0); }
}

void driveAzimut(int8_t sens, uint8_t vitesse) {
  if (sens > 0)      { ledcWrite(RPWM_AZ, vitesse); ledcWrite(LPWM_AZ, 0); }
  else if (sens < 0) { ledcWrite(RPWM_AZ, 0); ledcWrite(LPWM_AZ, vitesse); }
  else               { ledcWrite(RPWM_AZ, 0); ledcWrite(LPWM_AZ, 0); }
}

#define PIN_POT_CANAL_A     36
#define PIN_POT_CANAL_B     39
#define NB_ECHANTILLONS_POT  8

#define POT_A_0DEG    3918
#define POT_A_40DEG    269
#define POT_B_0DEG     137
#define POT_B_40DEG   3684
#define POT_ANGLE_MAX_DEG  40.0

int lireBrutFiltre(int pin) {
  long somme = 0;
  for (int i = 0; i < NB_ECHANTILLONS_POT; i++) { somme += analogRead(pin); delay(1); }
  return somme / NB_ECHANTILLONS_POT;
}

double brutVersDegres(int brut, int brut0Deg, int brut40Deg) {
  return (double)(brut - brut0Deg) * POT_ANGLE_MAX_DEG / (double)(brut40Deg - brut0Deg);
}

double lireInclinaisonReelle() {
  int brutA = lireBrutFiltre(PIN_POT_CANAL_A);
  int brutB = lireBrutFiltre(PIN_POT_CANAL_B);
  double degA = brutVersDegres(brutA, POT_A_0DEG, POT_A_40DEG);
  double degB = brutVersDegres(brutB, POT_B_0DEG, POT_B_40DEG);
  return (degA + degB) / 2.0;
}

double azimutEstimeDeg = 0.0;
double vitesseAzimutDegS = 4.0;
unsigned long lastAzimutUpdate = 0;
int8_t sensAzimutActuel = 0;

void updateEstimationAzimut() {
  unsigned long now = millis();
  double dt = (now - lastAzimutUpdate) / 1000.0;
  lastAzimutUpdate = now;
  if (sensAzimutActuel != 0) {
    azimutEstimeDeg += vitesseAzimutDegS * sensAzimutActuel * dt;
  }
}

// METEO EN LIGNE (Open-Meteo, gratuit, sans cle)
// ============================================================

float meteoTemperatureC   = 0;
float meteoVentKmh        = 0;
float meteoPressionHpa    = 0;
int   meteoProbaPluiePct  = 0;
bool  meteoValide          = false;
unsigned long meteoDerniereMaj = 0;
#define METEO_INTERVAL_MS  600000UL

void fetchMeteo() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = "http://api.open-meteo.com/v1/forecast?latitude=" + String(LATITUDE_DEG, 4) +
               "&longitude=" + String(LONGITUDE_DEG, 4) +
               "&current=temperature_2m,wind_speed_10m,surface_pressure" +
               "&hourly=precipitation_probability&forecast_hours=3&timezone=auto";

  http.begin(url);
  int code = http.GET();

  if (code == 200) {
    String payload = http.getString();
    JsonDocument doc;
    DeserializationError err = deserializeJson(doc, payload);
    if (!err) {
      meteoTemperatureC  = doc["current"]["temperature_2m"] | 0.0;
      meteoVentKmh       = doc["current"]["wind_speed_10m"] | 0.0;
      meteoPressionHpa   = doc["current"]["surface_pressure"] | 0.0;
      meteoProbaPluiePct = doc["hourly"]["precipitation_probability"][1] | 0;
      meteoValide = true;
      meteoDerniereMaj = millis();
      Serial.println("Meteo mise a jour.");
    } else {
      Serial.println("Erreur parsing JSON meteo.");
    }
  } else {
    Serial.printf("Erreur requete meteo, code HTTP: %d\n", code);
  }
  http.end();
}

void handleMonter() {
  retourAPlatActif = false;
  if (modeVentReduit && lireInclinaisonReelle() >= ANGLE_MAX_VENT_REDUIT_DEG) {
    etatVertical = "STOP";
    driveInclinaison(1, 0);
    Serial.println("MONTER refuse : amplitude limitee par vent modere (>=40 km/h).");
    server.sendHeader("Location", "/"); server.send(303);
    return;
  }
  etatVertical = "MONTER";
  driveInclinaison(1, modeVentReduit ? VITESSE_REDUITE : VITESSE_MANUELLE);
  arretAutoVerticalAt = millis() + DUREE_MAX_MS;
  Serial.println("Commande reçue : MONTER");
  server.sendHeader("Location", "/"); server.send(303);
}

void handleStatus() {
  double inclinaisonReelle = lireInclinaisonReelle();
  double azNorm = fmod(azimutEstimeDeg, 360.0);
  if (azNorm < 0) azNorm += 360.0;

  long secV = (arretAutoVerticalAt != 0) ? (long)((arretAutoVerticalAt - millis()) / 1000) : 0;
  long secH = (arretAutoHorizontalAt != 0) ? (long)((arretAutoHorizontalAt - millis()) / 1000) : 0;
  if (secV < 0) secV = 0;
  if (secH < 0) secH = 0;

  JsonDocument doc;
  doc["az"] = azNorm;
  doc["incl"] = inclinaisonReelle;
  doc["etatV"] = etatVertical;
  doc["etatH"] = etatHorizontal;
  doc["secV"] = secV;
  doc["secH"] = secH;
  doc["uptime"] = millis() / 1000;
  doc["rssi"] = WiFi.RSSI();
  doc["meteoValide"] = meteoValide;
  doc["temp"] = meteoTemperatureC;
  doc["vent"] = meteoVentKmh;
  doc["pression"] = meteoPressionHpa;
  doc["pluie"] = meteoProbaPluiePct;
  doc["meteoAge"] = meteoValide ? (millis() - meteoDerniereMaj) / 1000 : 0;

  doc["bmeOK"] = bmeOK;
  if (bmeOK) {
    doc["pressionLocale"] = bmp.readPressure() / 100.0;
    doc["temperatureLocale"] = bmp.readTemperature();
    doc["humiditeLocale"] = -1; // BMP280 ne mesure pas l'humidite
  }

  doc["rtcOK"] = rtcOK;
  doc["ventKmh"] = vitesseVentKmh;
  doc["ventRafaleKmh"] = vitesseRafaleKmh;
  doc["securiteVent"] = securiteVentActive;
  doc["modeVentReduit"] = modeVentReduit;
  doc["trend"] = tendanceVent;
  if (rtcOK) {
    DateTime now = rtc.now();
    char buf[20];
    snprintf(buf, sizeof(buf), "%04d-%02d-%02d %02d:%02d:%02d",
             now.year(), now.month(), now.day(), now.hour(), now.minute(), now.second());
    doc["heureRTC"] = buf;
  }

  String out;
  serializeJson(doc, out);
  server.send(200, "application/json", out);
}

// HISTORIQUE DES EVENEMENTS VENT (dernieres mises en securite)
// ============================================================
void handleHistoriqueVent() {
  JsonDocument doc;
  JsonArray arr = doc.to<JsonArray>();
  for (int i = 0; i < histRempli; i++) {
    int idx = (histIndex - histRempli + i + NB_EVENEMENTS_VENT) % NB_EVENEMENTS_VENT;
    JsonObject o = arr.add<JsonObject>();
    o["h"] = histHeure[idx];
    o["v"] = histVitesse[idx];
    o["c"] = histCause[idx];
  }
  String out;
  serializeJson(doc, out);
  server.send(200, "application/json", out);
}

// CONNEXION WIFI
// ============================================================

bool connectWiFi() {
  Serial.println("\nConnexion au WiFi...");
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  unsigned long start = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - start < 15000) {
    Serial.print(".");
    delay(300);
  }
  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.print("WiFi CONNECTÉ ! IP : ");
    Serial.println(WiFi.localIP());
    return true;
  }
  Serial.println("ÉCHEC WiFi");
  return false;
}

// ============================================================
// SETUP
// ============================================================

void setup() {
  Serial.begin(115200);
  delay(500);

  setupMoteurs();
  lastAzimutUpdate = millis();
  setupAnemometre();
  lastMesureVent = millis();
  setupCapteursI2C();

  // Charge l'historique du vent depuis la mémoire flash
  prefs.begin("tracker", false);
  chargerHistoriqueVent();

  if (!connectWiFi()) return;

  synchroniserRTCDepuisNTP();

  if (MDNS.begin("tracker")) {
    Serial.println("mDNS actif : accessible aussi via http://tracker.local");
  } else {
    Serial.println("mDNS non demarre (utilise l'adresse IP directement).");
  }

  fetchMeteo();
  
  // Configure le flashage Over-The-Air (OTA)
  setupOTA();

  server.on("/", handleRoot);
  server.on("/status", handleStatus);
  server.on("/api/historique-vent", handleHistoriqueVent);
  server.on("/monter", handleMonter);
  server.on("/stop_vertical", handleStopVertical);
  server.on("/descendre", handleDescendre);
  server.on("/retour_a_plat", handleRetourAPlat);
  server.on("/gauche", handleGauche);
  server.on("/stop_horizontal", handleStopHorizontal);
  server.on("/droite", handleDroite);
  server.begin();

  Serial.print("Serveur Web démarré ! Ouvre : http://");
  Serial.println(WiFi.localIP());
}

// ============================================================
// LOOP
// ============================================================

void loop() {
  server.handleClient();
  ArduinoOTA.handle();  // Traite les demandes de flashage OTA
  updateEstimationAzimut();

  if (arretAutoVerticalAt != 0 && millis() > arretAutoVerticalAt) {
    etatVertical = "STOP";
    driveInclinaison(1, 0);
    arretAutoVerticalAt = 0;
    Serial.println("Arret automatique (securite vertical)");
  }
  if (arretAutoHorizontalAt != 0 && millis() > arretAutoHorizontalAt) {
    etatHorizontal = "STOP";
    driveAzimut(1, 0);
    sensAzimutActuel = 0;
    arretAutoHorizontalAt = 0;
    Serial.println("Arret automatique (securite horizontal)");
  }

  static unsigned long lastRetourCheck = 0;
  if (retourAPlatActif && millis() - lastRetourCheck >= 200) {
    lastRetourCheck = millis();
    double actuel = lireInclinaisonReelle();
    double erreur = 0.0 - actuel;

    if (fabs(erreur) < RETOUR_A_PLAT_TOLERANCE_DEG) {
      driveInclinaison(1, 0);
      retourAPlatActif = false;
      etatVertical = "STOP (a plat)";
      Serial.println("Retour a plat termine.");
    } else {
      int8_t sens = (erreur > 0) ? 1 : -1;
      driveInclinaison(sens, RETOUR_A_PLAT_VITESSE);
    }
  }

  if (millis() - meteoDerniereMaj >= METEO_INTERVAL_MS) {
    fetchMeteo();
  }

  static unsigned long lastHeartbeat = 0;
  updateVent();
  updateRafale();

  // Securite vent : retour a plat automatique si vent trop fort
  if (securiteVentActive && !retourAPlatActif) {
    retourAPlatActif = true;
    etatVertical = "RETOUR A PLAT (VENT)";
    arretAutoVerticalAt = 0;
  }
  if (millis() - lastHeartbeat >= 60000) {
    lastHeartbeat = millis();
    Serial.printf("[vivant] uptime: %lus | IP: %s\n",
                  millis() / 1000, WiFi.localIP().toString().c_str());
  }
}
